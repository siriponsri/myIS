@echo off
setlocal
set "RESEARCH_ROOT=%~dp0.."
for %%I in ("%RESEARCH_ROOT%") do set "RESEARCH_ROOT=%%~fI"
if "%MYIS_MLFLOW_PORT%"=="" set "MYIS_MLFLOW_PORT=5000"
where uv >nul 2>&1 || (echo ERROR: uv is required.& exit /b 2)

uv run --no-sync python "%RESEARCH_ROOT%\scripts\bootstrap_mlflow.py" --repository-root "%RESEARCH_ROOT%"
if errorlevel 1 (
  echo ERROR: MLflow bootstrap failed.
  pause
  exit /b 1
)

start "myIS MLflow" /min powershell -NoProfile -ExecutionPolicy Bypass -Command ^
  "Set-Location -LiteralPath '%RESEARCH_ROOT%'; uv run --no-sync python dashboard/mlflow/readonly_app.py serve --repository-root '%RESEARCH_ROOT%' --port '%MYIS_MLFLOW_PORT%'"
timeout /t 3 /nobreak >nul
start "" "http://127.0.0.1:%MYIS_MLFLOW_PORT%"
endlocal
